// Approach 1
var lengthOfLastWord = function(s) {
    let n = s.length - 1;
    while (n >= 0 && s[n] === ' ') n--; // this is for trim from last

    let count = 0;
    while (n >= 0 && s[n] !== ' ') { // count
        count++;
        n--;
    }
    return count;
};

// Approach 2
var lengthOfLastWord = function(s) {
    let n = s.length - 1;
    let count = 0;
    while (n >= 0) {
        if (s[n] !== " ") {
            count++;
        } else if (count > 0) {
            break;
        }
        n--;
    }
    return count;
};

var findWordsContaining = function(words, x) {
    let arr = []
    for(let i=0;i<words.length;i++){
        if(words[i].includes(x)) arr.push(i)
    }
    return arr
};

3. 771 - Jewels and Stones
=> jewels = "aA", stones = "aAAbbbb" => 3
// Code
var numJewelsInStones = function(jewels, stones) {
    let jSet = new Set(jewels);
    let count = 0;
    for (let c of stones) {
        if (jSet.has(c)) count++;
    }
    return count;
};

4 3541 - Find Most Frequent Vowel and Consonant
=> "successes" => 6
// Code
var maxFreqSum = function (s) {
    let map = {}
    for (i = 0; i < s.length; i++) {
        if (!map[s[i]]) {
            map[s[i]] = 1
        } else {
            ++map[s[i]]
        }
    }

    let vowels = ['a', 'e', 'i', 'o', 'u']
    let maxVowels = 0
    let maxConsonant = 0
    for (let i = 0; i < s.length; i++) {
        if (vowels.includes(s[i])) {
            if (map[s[i]] > maxVowels) {
                maxVowels = map[s[i]]
            }
        }
        else {
            if (map[s[i]] > maxConsonant) {
                maxConsonant = map[s[i]]
            }
        }
    }
    return maxConsonant + maxVowels
};
