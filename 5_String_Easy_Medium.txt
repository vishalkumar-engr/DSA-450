// Approach 1
var lengthOfLastWord = function(s) {
    let n = s.length - 1;
    while (n >= 0 && s[n] === ' ') n--; // this is for trim from last

    let count = 0;
    while (n >= 0 && s[n] !== ' ') { // count
        count++;
        n--;
    }
    return count;
};

// Approach 2
var lengthOfLastWord = function(s) {
    let n = s.length - 1;
    let count = 0;
    while (n >= 0) {
        if (s[n] !== " ") {
            count++;
        } else if (count > 0) {
            break;
        }
        n--;
    }
    return count;
};

3. 771 - Jewels and Stones
=> jewels = "aA", stones = "aAAbbbb" => 3
// Code
var numJewelsInStones = function(jewels, stones) {
    let jSet = new Set(jewels);
    let count = 0;
    for (let c of stones) {
        if (jSet.has(c)) count++;
    }
    return count;
};

4 3541 - Find Most Frequent Vowel and Consonant
=> "successes" => 6
// Code
var maxFreqSum = function (s) {
    let map = {}
    for (i = 0; i < s.length; i++) {
        if (!map[s[i]]) {
            map[s[i]] = 1
        } else {
            ++map[s[i]]
        }
    }

    let vowels = ['a', 'e', 'i', 'o', 'u']
    let maxVowels = 0
    let maxConsonant = 0
    for (let i = 0; i < s.length; i++) {
        if (vowels.includes(s[i])) {
            if (map[s[i]] > maxVowels) {
                maxVowels = map[s[i]]
            }
        }
        else {
            if (map[s[i]] > maxConsonant) {
                maxConsonant = map[s[i]]
            }
        }
    }
    return maxConsonant + maxVowels
};

var balancedStringSplit = function(s) {
    let temp=0,count=0
    for(let i =0;i<s.length;i++){
        if(s[i] === 'R'){
            ++temp
        }else{
            --temp
        }
        if(temp==0){
            count++
        }
    }
    return count
};

6. 541 - Reverse String II
=> s = "abcdefg", k = 2 => "bacdfeg"
// Code
var reverseStr = function(s, k) {
    let sArr = []
    for(let i=0;i<s.length;i=i+k){
        let str = ''
        for(let j=i;j<i+k;j++){
            if(s[j]){
                str+=s[j]
            }
        }
        sArr.push(str)
    }
    for(let i=0;i<sArr.length;i=i+2){
        sArr[i] = sArr[i].split('').reverse().join('');
    }
    return sArr.join('')
};

// Approach 2
var reverseStr = function (s, k) {
      s = s.split("")
      for (let x = 0; x < s.length; x = x + (2 * k)) {
          let n = Math.min(k, s.length - x)
          let mid = Math.floor(n / 2)
          for (let i = 0; i < mid; i++) {
              let temp = s[x + i]
              s[x + i] = s[x + n - 1 - i]
              s[x + n - 1 - i] = temp
          }
      }
      return s.join("")
  }; 

7. 125 - Valid Palindrome
=> "A man, a plan, a canal: Panama" => "amanaplanacanalpanama" => true
// Code
var isPalindrome = function (s) {
s = s.toLowerCase();
let filteredString = "";
let rev = "";
for (let i = 0; i < s.length; i++) {
    if (
    (s[i].charCodeAt() >= "a".charCodeAt() && s[i].charCodeAt() <= "z".charCodeAt()) ||
    (s[i].charCodeAt() >= "0".charCodeAt() && s[i].charCodeAt() <= "9".charCodeAt())
    ) {
    filteredString = filteredString + s[i];
    rev = s[i] + rev;
    }
}
return filteredString === rev;
};

10. 242 - Valid Anagram
=> s = "anagram", t = "nagaram" => true
// Code
var isAnagram = function(s, t) {
      if (s.length !== t.length) return false;
      let map = {};
      for (let i = 0; i < s.length; i++) {
          if (!map[s[i]]) {
              map[s[i]] = 1;
          } else {
              ++map[s[i]];
          }
      }
      for (let i = 0; i < t.length; i++) {
          if (!map[t[i]] || map[t[i]] < 0) {
              return false;
          } else {
              --map[t[i]];
          }
      }
      return true;
  };

11. 205 - Isomorphic Strings
=> s = "egg", t = "add" => true
// Code
var isIsomorphic = function(s, t) {
      let mapSToT = {};
      let mapTToS = {};
      for (let i = 0; i < s.length; i++) {
          if (!mapSToT[s[i]] && !mapTToS[t[i]]) {
              mapSToT[s[i]] = t[i];
              mapTToS[t[i]] = s[i];
          } else if (mapTToS[t[i]] !== s[i] || mapSToT[s[i]] !== t[i]) {
              return false;
          }
      }
      return true;
  };

12. 49 - Group Anagrams
=> ["eat","tea","tan","ate","nat","bat"] => [["bat"],["nat","tan"],["ate","eat","tea"]]
// Code
var groupAnagrams = function(strs) {
      let map = {};
      for (let i = 0; i < strs.length; i++) {
          let freqArr = Array(26).fill(0);
          let s = strs[i];
          for (let j = 0; j < s.length; j++) {
              let index = s[j].charCodeAt(0) - 'a'.charCodeAt(0);
              freqArr[index]++;
          }
          let key = "";
          for (let k = 0; k < 26; k++) {
              key += "#" + freqArr[k];
          }
          if (!map[key]) {
              map[key] = [s];
          } else {
              map[key].push(s);
          }
      }
      return Object.values(map);
  };
// Brut Approach
var groupAnagrams = function(strs) {
      let map = {};
      for (let i = 0; i < strs.length; i++) {
          let sortedStr = strs[i].split("").sort().join(""); 
          if (!map[sortedStr]) {
              map[sortedStr] = [strs[i]];
          } else {
              map[sortedStr].push(strs[i]);
          }
      }
      return Object.values(map);
  };
